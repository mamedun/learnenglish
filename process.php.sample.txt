    $allowed = $config['cors_allowed_origins'] ?? [];

    foreach ($allowed as $a) {

    }
    // allow local dev
    if (strpos($origin, 'localhost') !== false || strpos($origin, '127.0.0.1') !== false) $isAllowed = true;
    if ($isAllowed) {
        header("Access-Control-Allow-Origin: $origin");
        header('Access-Control-Allow-Credentials: true');
    }
}

function sanitizeErrorMessage($msg){
    if (!is_string($msg)) return $msg;
    // Hide any internal server URLs/domains from user
    $patterns = [
        '/https?:\/\/[^\s]*ichsanlabs\.com[^\s]*/i' => '[AI Server]',
        '/https?:\/\/[^\s]*sulapfoto[^\s]*/i' => '[AI Server]',
        '/sg\d+\.ichsanlabs\.com/i' => '[AI Server]',
        '/api\.sulapfoto\.id/i' => '[AI Server]',
        '/sulapfoto\.com/i' => '[AI Server]',
        '/ichsanlabs\.com/i' => '[AI Server]',
    ];
    foreach ($patterns as $pat=>$rep){
        $msg = preg_replace($pat, $rep, $msg);
    }
    // If after sanitization still contains internal hint and it's an error about all servers, replace entirely
    if (stripos($msg, 'All public servers failed') !== false || stripos($msg, 'ichsanlabs') !== false || stripos($msg, 'sulapfoto') !== false){
        $msg = 'Server AI sedang sibuk, coba lagi beberapa saat. Jika terus gagal hubungi admin.';
    }
    return $msg;
}
function jsonResponse($data, $code = 200) {
    // Sanitize error field if present
    if (isset($data['error']) && is_string($data['error'])){
        $data['error'] = sanitizeErrorMessage($data['error']);
    }
    if (isset($data['message']) && is_string($data['message'])){
        // only sanitize if it looks like it leaks internal URL
        if (preg_match('/ichsanlabs|sulapfoto|sg\d+/i', $data['message'])){
            $data['message'] = sanitizeErrorMessage($data['message']);
        }
    }
    http_response_code($code);
    echo json_encode($data, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES);
    exit;
}

function getBearerToken() {
    $h = $_SERVER['HTTP_AUTHORIZATION'] ?? $_SERVER['REDIRECT_HTTP_AUTHORIZATION'] ?? '';
    if (!$h) {
        // fallback from apache_request_headers
        if (function_exists('getallheaders')) {
            $headers = getallheaders();
            $h = $headers['Authorization'] ?? $headers['authorization'] ?? '';
        }
    }
    if (preg_match('/Bearer\s+(.*)$/i', $h, $m)) return trim($m[1]);
    return null;
}

function db() {
    static $pdo = null;
    global $config;
    if ($pdo) return $pdo;
    $dbPath = $config['db_path'];
    $dir = dirname($dbPath);
    if (!is_dir($dir)) mkdir($dir, 0775, true);
    $pdo = new PDO('sqlite:' . $dbPath);
    $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
    $pdo->setAttribute(PDO::ATTR_DEFAULT_FETCH_MODE, PDO::FETCH_ASSOC);
    // init tables
    $pdo->exec("CREATE TABLE IF NOT EXISTS users (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        username TEXT UNIQUE NOT NULL,
        email TEXT,
        password_hash TEXT NOT NULL,
        tier TEXT NOT NULL DEFAULT 'micro',
        coins INTEGER NOT NULL DEFAULT 10,
        diamonds INTEGER NOT NULL DEFAULT 0,
        storage_limit_mb INTEGER NOT NULL DEFAULT 20,
        last_coin_refresh DATE,
        expires_at DATE,
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
        updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
    )");
    $pdo->exec("CREATE TABLE IF NOT EXISTS user_sessions (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        user_id INTEGER NOT NULL,
        token TEXT UNIQUE NOT NULL,
        device_info TEXT,
        ip TEXT,
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
        last_active DATETIME DEFAULT CURRENT_TIMESTAMP,
        expires_at DATETIME,
        FOREIGN KEY(user_id) REFERENCES users(id) ON DELETE CASCADE
    )");
    $pdo->exec("CREATE TABLE IF NOT EXISTS projects (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        user_id INTEGER NOT NULL,
        type TEXT NOT NULL, -- manual or auto
        title TEXT NOT NULL,
        topic TEXT,
        size_bytes INTEGER DEFAULT 0,
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
        updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
        FOREIGN KEY(user_id) REFERENCES users(id) ON DELETE CASCADE
    )");
    $pdo->exec("CREATE TABLE IF NOT EXISTS coin_logs (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        user_id INTEGER NOT NULL,
        amount INTEGER NOT NULL,
        reason TEXT,
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
        FOREIGN KEY(user_id) REFERENCES users(id) ON DELETE CASCADE
    )");
    $pdo->exec("CREATE TABLE IF NOT EXISTS diamond_logs (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        user_id INTEGER NOT NULL,
        amount INTEGER NOT NULL,
        reason TEXT,
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
        FOREIGN KEY(user_id) REFERENCES users(id) ON DELETE CASCADE
    )");
    $pdo->exec("CREATE TABLE IF NOT EXISTS app_config (
        key TEXT PRIMARY KEY,
        value TEXT
    )");

    // migrations: add expires_at if missing
    try {
        $cols = $pdo->query("PRAGMA table_info(users)")->fetchAll();
        $hasExpires = false;
        $hasLastRefresh = false;
        foreach ($cols as $c) {
            if ($c['name']==='expires_at') $hasExpires=true;
            if ($c['name']==='last_coin_refresh') $hasLastRefresh=true;
        }
        if (!$hasExpires) $pdo->exec("ALTER TABLE users ADD COLUMN expires_at DATE");
        if (!$hasLastRefresh) $pdo->exec("ALTER TABLE users ADD COLUMN last_coin_refresh DATE");
    } catch (Exception $e) {}

    // migrate free -> micro
    try {
        $pdo->exec("UPDATE users SET tier='micro' WHERE tier='free'");
    } catch (Exception $e) {}

    // ensure admin exists
    $stmt = $pdo->query("SELECT COUNT(*) as c FROM users WHERE tier='admin'");
    $row = $stmt->fetch();
    if (($row['c'] ?? 0) == 0) {
        $admin = $config['admin_default'];
        $hash = password_hash($admin['password'], PASSWORD_BCRYPT);
        $pdo->prepare("INSERT INTO users (username,email,password_hash,tier,coins,diamonds,storage_limit_mb,last_coin_refresh,expires_at) VALUES (?,?,?,?,?,?,?,date('now'),date('now','+10 years'))")
            ->execute([$admin['username'], $admin['email'], $hash, 'admin', -1, 9999, 1024]);
    }
    return $pdo;
}

function getPrivateConfig() {
    global $config;
    $filePriv = $config['private'] ?? [];
    try {
        $pdo = db();
        $stmt = $pdo->prepare("SELECT value FROM app_config WHERE key='private_config'");
        $stmt->execute();
        $row = $stmt->fetch();
        if ($row && !empty($row['value'])) {
            $dbVal = json_decode($row['value'], true);
            if (is_array($dbVal)) {
                $filePriv = array_merge($filePriv, $dbVal);
            } else {
                $inner = json_decode($dbVal['value'] ?? '', true);
                if (is_array($inner)) $filePriv = array_merge($filePriv, $inner);
            }
        }
    } catch (Exception $e) {}

    // UNIFY: migrate old text_model_lite -> text_model if exists
    if (!empty($filePriv['text_model_lite'])) {
        // if text_model still old (2.0-flash) and lite exists, prefer lite
        if (empty($filePriv['text_model']) || $filePriv['text_model']==='gemini-2.0-flash' || stripos($filePriv['text_model'],'imagen')!==false) {
            $filePriv['text_model'] = $filePriv['text_model_lite'];
        }
        unset($filePriv['text_model_lite']); // hapus biar unify
    }
    if (empty($filePriv['text_model'])) {
        $filePriv['text_model'] = 'gemini-3.1-flash-lite';
    }
    if (empty($filePriv['image_model'])) {
        $filePriv['image_model'] = 'gemini-3.1-flash-image';
    }
    if (empty($filePriv['image_models'])) {
        $filePriv['image_models'] = [
            'nano_banana' => 'gemini-2.5-flash-image-preview',
            'nano_banana_2' => 'gemini-3.1-flash-image',
            'nano_banana_2_lite' => 'gemini-3.1-flash-lite-image'
        ];
    }
    // Enforce 1K 16:9 untuk hemat biaya (sesuai request user)
    if (empty($filePriv['image_size'])) $filePriv['image_size']='1K';
    if (empty($filePriv['image_aspect'])) $filePriv['image_aspect']='16:9';

    return $filePriv;
}

function isUserExpired($userRow) {
    if (($userRow['tier'] ?? '') === 'admin') return false;
    $exp = $userRow['expires_at'] ?? null;
    if (!$exp) return false; // no expiry set = not expired
    $today = date('Y-m-d');
    return $exp < $today;
}

function getUserByToken($token) {
    if (!$token) return null;
    $pdo = db();
    $stmt = $pdo->prepare("SELECT us.*, u.* FROM user_sessions us JOIN users u ON u.id=us.user_id WHERE us.token=? AND (us.expires_at IS NULL OR us.expires_at > datetime('now'))");
    $stmt->execute([$token]);
    $row = $stmt->fetch();
    if (!$row) return null;
    // update last_active
    $pdo->prepare("UPDATE user_sessions SET last_active=datetime('now') WHERE token=?")->execute([$token]);

    // daily coin refresh with cap logic
    $today = date('Y-m-d');
    if (($row['last_coin_refresh'] ?? '') !== $today && $row['tier'] !== 'admin') {
        $tierCfg = getTierConfig($row['tier']);
        $daily = $tierCfg['coins_daily'] ?? 10;
        $max = $tierCfg['coins_max'] ?? 15;
        $currentCoins = intval($row['coins'] ?? 0);
        // unlimited
        if ($daily === -1 || $max === -1) {
            // no refresh needed, but update date
            $pdo->prepare("UPDATE users SET last_coin_refresh=date('now'), updated_at=datetime('now') WHERE id=?")->execute([$row['user_id']]);
        } else {
            // only add if below cap
            if ($currentCoins < $max) {
                $newCoins = min($max, $currentCoins + $daily);
                $pdo->prepare("UPDATE users SET coins=?, last_coin_refresh=date('now'), updated_at=datetime('now') WHERE id=?")->execute([$newCoins, $row['user_id']]);
                $row['coins'] = $newCoins;
            } else {
                // already at or above cap (admin may have set higher), do not reduce, just update refresh date
                $pdo->prepare("UPDATE users SET last_coin_refresh=date('now'), updated_at=datetime('now') WHERE id=?")->execute([$row['user_id']]);
            }
        }
        // also handle diamond monthly for VIP if expired? We'll handle separately in a daily cron? For now check if tier vip and expires still valid, add monthly diamond on first day of month? Simpler: if tier vip and last refresh month changed, add 50? We'll implement monthly check: if diamond_monthly >0 and last refresh != today, and day == 1 or month changed, we add. For simplicity, we add diamond monthly on each coin refresh if tier has diamond_monthly and last refresh month != current month
        $tierCfg = getTierConfig($row['tier']);
        if (($tierCfg['diamond_monthly'] ?? 0) > 0) {
            // check if last_coin_refresh month != today month, or we can check if we already gave this month via a separate log? Simplified: if last refresh date's month != today month, give diamond
            $last = $row['last_coin_refresh'] ?? '';
            $lastMonth = substr($last, 0, 7);
            $curMonth = substr($today, 0, 7);
            if ($lastMonth !== $curMonth) {
                $addDiamond = intval($tierCfg['diamond_monthly']);
                $pdo->prepare("UPDATE users SET diamonds=diamonds+?, updated_at=datetime('now') WHERE id=?")->execute([$addDiamond, $row['user_id']]);
                $pdo->prepare("INSERT INTO diamond_logs (user_id, amount, reason) VALUES (?,?,?)")->execute([$row['user_id'], $addDiamond, 'monthly vip bonus']);
            }
        }
    }
    return $row;
}

function requireAuth() {
    $token = getBearerToken();
    $user = getUserByToken($token);
    if (!$user) jsonResponse(['success'=>false,'error'=>'unauthorized'], 401);
    return $user;
}

}

function getInput() {

function calcUserStorage($userId) {
    global $config;
    $base = rtrim($config['uploads_dir'], '/') . "/$userId";
    if (!is_dir($base)) return 0;
    $size = 0;
    $it = new RecursiveIteratorIterator(new RecursiveDirectoryIterator($base, FilesystemIterator::SKIP_DOTS));
    foreach ($it as $file) {
        if ($file->isFile()) $size += $file->getSize();
    }
    return $size;
}

function base64url_encode_php($data) {
    return rtrim(strtr(base64_encode($data), '+/', '-_'), '=');
}
function generatePublicJWT() {
    global $config;
    $auth = $config['public_auth'] ?? [];
    $secret = $auth['jwt_secret'] ?? 'sufo888';
    $sub = $auth['sub'] ?? 'api-client';
    $ttlMin = intval($auth['ttl_min'] ?? 30);
    $header = ['alg'=>'HS256','typ'=>'JWT'];
    $now = time();
    // leeway 60s for clock skew like TokenManager refreshLead
    $payload = ['sub'=>$sub,'iat'=>$now - 60,'exp'=>$now + $ttlMin*60];
    $h = base64url_encode_php(json_encode($header, JSON_UNESCAPED_SLASHES));
    $p = base64url_encode_php(json_encode($payload, JSON_UNESCAPED_SLASHES));
    $sig = hash_hmac('sha256', "$h.$p", $secret, true);
    $sigB64 = base64url_encode_php($sig);
    return "$h.$p.$sigB64";
}
function buildCurlHeaders($apiKey, $jwt){
    $headers = [];
    if ($apiKey) $headers[] = "X-API-Key: $apiKey";
    if ($jwt) $headers[] = "Authorization: Bearer $jwt";
    return $headers;
}
function applyRobustCurlOptions($ch){
    // Mirror browser resilience: ignore SSL CA issues, force IPv4, add UA, follow redirects, handle gzip
    // Use browser-like UA to avoid WAF blocking datacenter UA
    curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, false);
    curl_setopt($ch, CURLOPT_SSL_VERIFYHOST, false);
    curl_setopt($ch, CURLOPT_IPRESOLVE, CURL_IPRESOLVE_V4);
    curl_setopt($ch, CURLOPT_FOLLOWLOCATION, true);
    curl_setopt($ch, CURLOPT_ENCODING, '');
    curl_setopt($ch, CURLOPT_USERAGENT, 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/122.0.0.0 Safari/537.36');
    curl_setopt($ch, CURLOPT_HTTP_VERSION, CURL_HTTP_VERSION_1_1);
    // Force TLS 1.2+ like browser
    if (defined('CURL_SSLVERSION_TLSv1_2')) curl_setopt($ch, CURLOPT_SSLVERSION, CURL_SSLVERSION_TLSv1_2);
    // timeouts set outside
}
function buildBrowserHeaders($apiKey, $jwt){
    $h = [];
    if ($apiKey) $h[] = "X-API-Key: $apiKey";
    if ($jwt) $h[] = "Authorization: Bearer $jwt";
    $h[] = "Accept: application/json, text/plain, */*";
    $h[] = "Accept-Language: en-US,en;q=0.9,id;q=0.8";
    $h[] = "Cache-Control: no-cache";
    $h[] = "Pragma: no-cache";
    $h[] = "Origin: https://www.xpcore.net";
    $h[] = "Referer: https://www.xpcore.net/";
    return $h;
}


function proxyToPublicServer($endpoint, $payload, $isFormData = false) {
    global $config;
    $servers = $config['public_servers'] ?? [];
    shuffle($servers);
    $path = $config['endpoints'][$endpoint] ?? "/$endpoint";

    $auth = $config['public_auth'] ?? [];
    $apiKey = $auth['api_key'] ?? 'kalaubutuhapikeysulapfotohubungiadminmurahdanunlimited';
    $jwt = generatePublicJWT();

    // OPTIMIZED: Use curl_multi parallel like multi-app.html Promise.all, return first success (fastest)
    // Previously sequential loop caused 9*10s timeout = very slow on hosting with firewall
    $mh = curl_multi_init();
    $handles = [];
    $tmpFilesToCleanup = []; // keep refs for FormData temp files if needed (already handled outside)

    foreach ($servers as $i=>$base){
        $url = rtrim($base, '/') . $path;
        $ch = curl_init($url);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_TIMEOUT, 30); // reduced from 75 to 30 for faster failover (multi-app uses 8s)
        curl_setopt($ch, CURLOPT_CONNECTTIMEOUT, 5); // reduced from 10 to 5
        applyRobustCurlOptions($ch);
        $headers = buildBrowserHeaders($apiKey, $jwt);
        if ($isFormData) {
            curl_setopt($ch, CURLOPT_POST, true);
            curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
            curl_setopt($ch, CURLOPT_HTTPHEADER, $headers);
        } else {
            curl_setopt($ch, CURLOPT_POST, true);
            curl_setopt($ch, CURLOPT_POSTFIELDS, is_string($payload) ? $payload : json_encode($payload, JSON_UNESCAPED_SLASHES));
            $headers[] = 'Content-Type: application/json';
            curl_setopt($ch, CURLOPT_HTTPHEADER, $headers);
        }
        curl_multi_add_handle($mh, $ch);
        $handles[$i] = ['ch'=>$ch, 'base'=>$base, 'url'=>$url];
    }

    $running = null;
    $firstSuccess = null;
    $lastError = null;
    $lastCode = 0;

    // Execute parallel, but with early exit when first success found (like Promise.race)
    do {
        $status = curl_multi_exec($mh, $running);
        // Check for completed handles
        while ($info = curl_multi_info_read($mh)) {
            $ch = $info['handle'];
            $idx = null;
            foreach ($handles as $k=>$h) { if ($h['ch']===$ch) { $idx=$k; break; } }
            if ($idx===null) continue;
            $resp = curl_multi_getcontent($ch);
            $code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
            $err = curl_error($ch);
            $base = $handles[$idx]['base'];

            if ($resp !== false && $code >=200 && $code <300) {
                $decoded = json_decode($resp, true);
                if (json_last_error()===JSON_ERROR_NONE) {
                    $firstSuccess = $decoded;
                    break 2; // break both loops - we have success
                }
                if (!empty($resp)) {
                    $firstSuccess = ['success'=>true,'raw'=>$resp,'imageUrl'=>$resp];
                    break 2;
                }
            }
            $lastError = $err ?: "HTTP $code";
            $lastCode = $code;
            error_log("[proxyToPublicServer-parallel] $endpoint failed base=".substr($base,0,40)." code=$code err=$err");
        }
        if ($firstSuccess) break;
        // Prevent busy loop
        if ($running) curl_multi_select($mh, 0.5);
    } while ($running > 0);

    // Cleanup
    foreach ($handles as $h) {
        curl_multi_remove_handle($mh, $h['ch']);
        curl_close($h['ch']);
    }
    curl_multi_close($mh);

    if ($firstSuccess) return $firstSuccess;

    // If parallel all failed, fallback to sequential retry with fresh JWT (old logic as last resort)
    $jwt2 = generatePublicJWT();
    foreach ($servers as $base) {
        $url = rtrim($base, '/') . $path;
        $ch = curl_init($url);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_TIMEOUT, 20);
        curl_setopt($ch, CURLOPT_CONNECTTIMEOUT, 5);
        applyRobustCurlOptions($ch);
        $headers = buildBrowserHeaders($apiKey, $jwt2);
        if ($isFormData) {
            curl_setopt($ch, CURLOPT_POST, true);
            curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
            curl_setopt($ch, CURLOPT_HTTPHEADER, $headers);
        } else {
            curl_setopt($ch, CURLOPT_POST, true);
            curl_setopt($ch, CURLOPT_POSTFIELDS, is_string($payload) ? $payload : json_encode($payload, JSON_UNESCAPED_SLASHES));
            $headers[] = 'Content-Type: application/json';
            curl_setopt($ch, CURLOPT_HTTPHEADER, $headers);
        }
        $resp = curl_exec($ch);
        $code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        $err = curl_error($ch);
        curl_close($ch);
        if ($resp !== false && $code >=200 && $code <300) {
            $decoded = json_decode($resp, true);
            if (json_last_error()===JSON_ERROR_NONE) return $decoded;
            if (!empty($resp)) return ['success'=>true,'raw'=>$resp,'imageUrl'=>$resp];
        }
        $lastError = $err ?: "HTTP $code";
        $lastCode = $code;
    }

    $msg = "Server AI sedang sibuk, coba lagi beberapa saat. Jika terus gagal hubungi admin.";
    if ($lastCode === 401 || $lastCode === 403) {
        $msg = "Server AI sedang sibuk (auth gagal), coba lagi. Jika terus gagal hubungi admin.";
    } else if ($lastCode==0 || $lastCode>=500){
        $msg = "Koneksi ke Server AI terputus, coba lagi beberapa saat. (Hosting firewall mungkin block outbound, tapi client-side check di Admin Panel bisa tetap ONLINE)";
    }
    return ['success'=>false,'error'=>$msg,'code'=>$lastCode,'debug'=>$lastError];
}

function proxyToPrivateGemini($prompt, $images = [], $mode = 'chat', $options = []) {
    $priv = getPrivateConfig();
    if (empty($priv['enabled']) || empty($priv['api_key'])) {
        return ['success'=>false,'error'=>'Private server not configured'];
    }
    $apiKey = $priv['api_key'];
    $base = rtrim($priv['base_url'] ?? 'https://generativelanguage.googleapis.com', '/');

    // Determine model with override support - UNIFIED: text_model is already lite (hemat)
    if ($mode === 'generate') {
        $model = $options['model'] ?? $priv['image_model'] ?? 'gemini-3.1-flash-image';
    } else {
        $model = $options['model'] ?? $priv['text_model'] ?? 'gemini-3.1-flash-lite';
    }

    // === TEXT / CHAT MODE ===
    if ($mode === 'chat') {
        $url = "$base/v1beta/models/$model:generateContent?key=$apiKey";
        $parts = [['text'=>$prompt]];
        foreach ($images as $imgDataUrl) {
            if (preg_match('/data:(.*?);base64,(.*)/', $imgDataUrl, $m)) {
                $parts[] = ['inline_data'=>['mime_type'=>$m[1],'data'=>$m[2]]];
            }
        }
        $payloadArr = ['contents'=>[['parts'=>$parts]]];

        // Optional response_schema for JSON output (cost saving + accuracy)
        if (!empty($options['response_schema'])) {
            $schema = $options['response_schema'];
            if (is_string($schema)) {
                $decodedSchema = json_decode($schema, true);
                if ($decodedSchema) $schema = $decodedSchema;
            }
            $payloadArr['generationConfig'] = [
                'responseMimeType' => 'application/json',
                'responseSchema' => $schema,
                'temperature' => $options['temperature'] ?? 0.2,
                'maxOutputTokens' => $options['max_tokens'] ?? 8192
            ];
        } else {
            // cost efficient defaults for analysis
            $payloadArr['generationConfig'] = [
                'temperature' => $options['temperature'] ?? 0.3,
                'maxOutputTokens' => $options['max_tokens'] ?? 8192
            ];
            // if prompt asks for JSON, hint model to output JSON only
            if (stripos($prompt, 'Return JSON') !== false || stripos($prompt, 'JSON') !== false) {
                $payloadArr['generationConfig']['responseMimeType'] = 'application/json';
            }
        }

        $payload = json_encode($payloadArr, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES);
        $ch = curl_init($url);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_POST, true);
        curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
        curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
        curl_setopt($ch, CURLOPT_TIMEOUT, 90);
        $resp = curl_exec($ch);
        $code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        $err = curl_error($ch);
        curl_close($ch);
        $decoded = json_decode($resp, true);
        if (isset($decoded['candidates'][0]['content']['parts'][0]['text'])) {
            return ['success'=>true,'response'=>$decoded['candidates'][0]['content']['parts'][0]['text'], 'model'=>$model, 'raw'=>$decoded];
        }
        // If schema was used, sometimes text is still in candidates
        if ($decoded) {
            // try to extract text anyway
            if (isset($decoded['candidates'])) {
                return ['success'=>false,'error'=>'Gemini returned no text','raw'=>$decoded,'code'=>$code,'model'=>$model];
            }
            return $decoded;
        }
        return ['success'=>false,'error'=>"Gemini text error $code: $resp err:$err", 'model'=>$model];
    }

    // === IMAGE GENERATION MODE ===
    // Try Interactions API for nano-banana models
    $isNanoBanana = (stripos($model, 'flash-image') !== false || stripos($model, 'image-preview') !== false || stripos($model, 'flash-lite-image') !== false);

    if ($isNanoBanana) {
        // Use new Interactions API (cost efficient, supports 16:9)
        $url = "$base/v1beta/interactions?key=$apiKey";
        // Alternative endpoint: /v1beta/models/{model}:generateContent with responseModalities IMAGE also works for 2.5
        // We'll try interactions first

        $inputBlocks = [];
        $inputBlocks[] = ['type'=>'text','text'=>$prompt];
        foreach ($images as $imgDataUrl) {
            if (preg_match('/data:(.*?);base64,(.*)/', $imgDataUrl, $m)) {
                $mime = $m[1];
                $b64 = $m[2];
                $inputBlocks[] = ['type'=>'image','mime_type'=>$mime,'data'=>$b64];
            }
        }

        // ENFORCED: hasil image hanya 1K max dengan 16:9 untuk hemat biaya (sesuai request)
        $aspect = '16:9'; // paksa 16:9
        $imageSize = '1K'; // paksa 1K max, bukan 2K
        // allow override only if explicitly allowed, but default enforced
        if (!empty($priv['image_aspect'])) $aspect = $priv['image_aspect'];
        if (!empty($priv['image_size'])) $imageSize = $priv['image_size'];
        // force 16:9 & 1K regardless of frontend request untuk private server
        $aspect = '16:9';
        $imageSize = '1K';

        $payloadArr = [
            'model' => $model,
            'input' => $inputBlocks,
            'response_format' => [
                'type' => 'image',
                'aspect_ratio' => $aspect,
                'image_size' => $imageSize
            ]
        ];

        $payload = json_encode($payloadArr, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES);
        $ch = curl_init($url);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_POST, true);
        curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
        curl_setopt($ch, CURLOPT_HTTPHEADER, [
            'Content-Type: application/json',
            'x-goog-api-key: '.$apiKey,
            'Api-Revision: 2026-05-20'
        ]);
        curl_setopt($ch, CURLOPT_TIMEOUT, 120);
        $resp = curl_exec($ch);
        $code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        $err = curl_error($ch);
        curl_close($ch);
        $decoded = json_decode($resp, true);

        if ($decoded) {
            // Parse output_image
            $imageData = null;
            $mime = 'image/png';
            if (isset($decoded['output_image']['data'])) {
                $imageData = $decoded['output_image']['data'];
                $mime = $decoded['output_image']['mime_type'] ?? 'image/png';
            } elseif (isset($decoded['output_images'][0]['data'])) {
                $imageData = $decoded['output_images'][0]['data'];
                $mime = $decoded['output_images'][0]['mime_type'] ?? 'image/png';
            } elseif (isset($decoded['steps'])) {
                foreach ($decoded['steps'] as $step) {
                    if (($step['type'] ?? '') === 'model_output' && isset($step['content'])) {
                        foreach ($step['content'] as $block) {
                            if (($block['type'] ?? '') === 'image' && isset($block['data'])) {
                                $imageData = $block['data'];
                                $mime = $block['mime_type'] ?? 'image/png';
                                break 2;
                            }
                        }
                    }
                }
            } elseif (isset($decoded['candidates'][0]['content']['parts'])) {
                // fallback generateContent style
                foreach ($decoded['candidates'][0]['content']['parts'] as $part) {
                    if (isset($part['inlineData']['data'])) {
                        $imageData = $part['inlineData']['data'];
                        $mime = $part['inlineData']['mimeType'] ?? 'image/png';
                        break;
                    }
                    if (isset($part['inline_data']['data'])) {
                        $imageData = $part['inline_data']['data'];
                        $mime = $part['inline_data']['mime_type'] ?? 'image/png';
                        break;
                    }
                }
            }

            if ($imageData) {
                $dataUrl = "data:$mime;base64,$imageData";
                return ['success'=>true,'imageUrl'=>$dataUrl,'model'=>$model,'raw'=>$decoded];
            }
            // If no image but has text error, return error
            if (isset($decoded['error'])) {
                return ['success'=>false,'error'=>'Gemini image error: '.json_encode($decoded['error']),'code'=>$code,'model'=>$model,'raw'=>$decoded];
            }
            // If we got here, try fallback to generateContent method
        }

        // Fallback: try generateContent with IMAGE modality (works for gemini-2.5-flash-image-preview)
        $url2 = "$base/v1beta/models/$model:generateContent?key=$apiKey";
        $parts = [['text'=>$prompt]];
        foreach ($images as $imgDataUrl) {
            if (preg_match('/data:(.*?);base64,(.*)/', $imgDataUrl, $m)) {
                $parts[] = ['inline_data'=>['mime_type'=>$m[1],'data'=>$m[2]]];
            }
        }
        $payloadArr2 = [
            'contents'=>[['parts'=>$parts]],
            'generationConfig'=>[
                'responseModalities'=>['IMAGE','TEXT'],
                'imageConfig'=>[
                    'aspectRatio'=>$aspect
                ]
            ]
        ];
        $payload2 = json_encode($payloadArr2, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES);
        $ch2 = curl_init($url2);
        curl_setopt($ch2, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch2, CURLOPT_POST, true);
        curl_setopt($ch2, CURLOPT_POSTFIELDS, $payload2);
        curl_setopt($ch2, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
        curl_setopt($ch2, CURLOPT_TIMEOUT, 120);
        $resp2 = curl_exec($ch2);
        $code2 = curl_getinfo($ch2, CURLINFO_HTTP_CODE);
        $err2 = curl_error($ch2);
        curl_close($ch2);
        $decoded2 = json_decode($resp2, true);
        if ($decoded2 && isset($decoded2['candidates'][0]['content']['parts'])) {
            foreach ($decoded2['candidates'][0]['content']['parts'] as $part) {
                if (isset($part['inlineData']['data'])) {
                    $mime = $part['inlineData']['mimeType'] ?? 'image/png';
                    $dataUrl = "data:$mime;base64,".$part['inlineData']['data'];
                    return ['success'=>true,'imageUrl'=>$dataUrl,'model'=>$model,'raw'=>$decoded2];
                }
                if (isset($part['inline_data']['data'])) {
                    $mime = $part['inline_data']['mime_type'] ?? 'image/png';
                    $dataUrl = "data:$mime;base64,".$part['inline_data']['data'];
                    return ['success'=>true,'imageUrl'=>$dataUrl,'model'=>$model,'raw'=>$decoded2];
                }
            }
        }
        return ['success'=>false,'error'=>"Gemini image error fallback $code2: ".substr($resp2,0,500)." err:$err2",'model'=>$model];
    } else {
        // Legacy Imagen model via predict
        $url = "$base/v1beta/models/$model:predict?key=$apiKey";
        $instances = [['prompt'=>$prompt]];
        $payload = json_encode(['instances'=>$instances]);
        $ch = curl_init($url);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_POST, true);
        curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
        curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
        curl_setopt($ch, CURLOPT_TIMEOUT, 90);
        $resp = curl_exec($ch);
        $code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);
        $decoded = json_decode($resp, true);
        if ($decoded) {
            // Try to extract image
            if (isset($decoded['predictions'][0]['bytesBase64Encoded'])) {
                $b64 = $decoded['predictions'][0]['bytesBase64Encoded'];
                return ['success'=>true,'imageUrl'=>"data:image/png;base64,$b64",'model'=>$model,'raw'=>$decoded];
            }
            return $decoded;
        }
        return ['success'=>false,'error'=>"Gemini image error $code: $resp",'model'=>$model];
    }
}

// Routing
$action = $_GET['action'] ?? $_POST['action'] ?? '';
if (!$action) {
    $jsonInput = getInput();
    $action = $jsonInput['action'] ?? '';
}
$action = trim($action);
$input = getInput();

// For multipart/form-data, $_POST may contain action, but files in $_FILES
if (empty($action) && isset($_GET['action'])) $action = $_GET['action'];

switch ($action) {
    // Public registration catalogue: explicit allowlist, never expose admin_config_get.
    case 'registration_info': {
        header('Cache-Control: no-store');
        $pdo = db();
        $stmt = $pdo->prepare("SELECT value FROM app_config WHERE key='wa_admin'");
        $stmt->execute();
        $stored = $stmt->fetchColumn();
        $wa = $stored !== false ? (json_decode($stored, true) ?? $stored) : ($config['wa_admin'] ?? '');
        $wa = is_scalar($wa) ? preg_replace('/[^0-9]/', '', (string)$wa) : '';
        if (substr($wa, 0, 1) === '0') $wa = '62'.substr($wa, 1);
        if (!preg_match('/^[1-9][0-9]{7,14}$/', $wa)) $wa = '';
        $plans = [];
        $publicFields = ['label','price','price_yearly_total','coins_daily','storage_mb','max_devices','diamond_bonus','diamond_monthly','is_custom'];
        foreach (['micro','regular','premium','vip','organization'] as $tier) {
            $tierConfig = getTierConfig($tier);
            $plan = ['id'=>$tier];
            foreach ($publicFields as $field) {
                if (array_key_exists($field, $tierConfig)) $plan[$field] = $tierConfig[$field];
            }
            $plans[] = $plan;
        }
        jsonResponse(['success'=>true, 'wa_admin'=>$wa, 'plans'=>$plans]);
        break;
    }
    case 'login': {
        $username = trim($input['username'] ?? $_POST['username'] ?? '');
        $password = $input['password'] ?? $_POST['password'] ?? '';
        $device = substr($input['device'] ?? $_POST['device'] ?? ($_SERVER['HTTP_USER_AGENT'] ?? 'unknown'), 0, 200);
        if (!$username || !$password) jsonResponse(['success'=>false,'error'=>'Username & password required'], 400);
        $pdo = db();
        $stmt = $pdo->prepare("SELECT * FROM users WHERE username=?");
        $stmt->execute([$username]);
        $user = $stmt->fetch();
        if (!$user || !password_verify($password, $user['password_hash'])) {
            jsonResponse(['success'=>false,'error'=>'Invalid credentials'], 401);
        }
        // check expiry - still allow login but flag expired
        $isExpired = isUserExpired($user);

        // device limit per tier - HYBRID with confirmation
        $tierCfg = getTierConfig($user['tier']);
        $maxDevices = $tierCfg['max_devices'] ?? $config['auth']['max_devices_default'] ?? 1;
        $force = !empty($input['force']) || !empty($input['force_logout']) || !empty($_POST['force']) || !empty($_POST['force_logout']);
        $stmt = $pdo->prepare("SELECT COUNT(*) as c FROM user_sessions WHERE user_id=? AND (expires_at IS NULL OR expires_at > datetime('now'))");
        $stmt->execute([$user['id']]);
        $count = $stmt->fetch()['c'] ?? 0;
        if ($count >= $maxDevices) {
            if ($maxDevices == 1) {
                // for single device, auto-kick oldest (allow login but replace) - no confirmation needed
                $pdo->prepare("DELETE FROM user_sessions WHERE id IN (SELECT id FROM user_sessions WHERE user_id=? ORDER BY last_active ASC LIMIT 1)")->execute([$user['id']]);
            } else {
                // for multi device (2, 50, 100), if not force, return device list + error for confirmation
                if (!$force) {
                    $stmt2 = $pdo->prepare("SELECT id, device_info, ip, created_at, last_active FROM user_sessions WHERE user_id=? AND (expires_at IS NULL OR expires_at > datetime('now')) ORDER BY last_active DESC");
                    $stmt2->execute([$user['id']]);
                    $active = $stmt2->fetchAll();
                    jsonResponse([
                        'success'=>false,
                        'error'=>'device_limit',
                        'message'=>"Maksimal $maxDevices device login bersamaan untuk tier {$user['tier']}. Anda sudah login di $count device. Logout device lain dulu atau force logout tertua?",
                        'max_devices'=>$maxDevices,
                        'current_count'=>$count,
                        'active_sessions'=>$active,
                        'need_confirmation'=>true
                    ], 403);
                } else {
                    // force=1 -> kick oldest
                    $pdo->prepare("DELETE FROM user_sessions WHERE id IN (SELECT id FROM user_sessions WHERE user_id=? ORDER BY last_active ASC LIMIT 1)")->execute([$user['id']]);
                }
            }
        }
        $token = bin2hex(random_bytes(32));
        $ttlHours = $config['auth']['token_ttl_hours'] ?? 168;
        $pdo->prepare("INSERT INTO user_sessions (user_id, token, device_info, ip, expires_at) VALUES (?,?,?,?,datetime('now','+{$ttlHours} hours'))")
            ->execute([$user['id'], $token, $device, $_SERVER['REMOTE_ADDR'] ?? '']);

        // refresh coins with cap logic (same as getUserByToken)
        $today = date('Y-m-d');
        if (($user['last_coin_refresh'] ?? '') !== $today && $user['tier'] !== 'admin') {
            $daily = $tierCfg['coins_daily'] ?? 10;
            $max = $tierCfg['coins_max'] ?? 15;
            $cur = intval($user['coins'] ?? 0);
            if ($daily !== -1 && $max !== -1) {
                if ($cur < $max) {
                    $newCoins = min($max, $cur + $daily);
                    $pdo->prepare("UPDATE users SET coins=?, last_coin_refresh=date('now'), updated_at=datetime('now') WHERE id=?")->execute([$newCoins, $user['id']]);
                    $user['coins'] = $newCoins;
                } else {
                    $pdo->prepare("UPDATE users SET last_coin_refresh=date('now') WHERE id=?")->execute([$user['id']]);
                }
            } else {
                $pdo->prepare("UPDATE users SET last_coin_refresh=date('now') WHERE id=?")->execute([$user['id']]);
            }
        }

        unset($user['password_hash']);
        $user['is_expired'] = $isExpired;
        jsonResponse(['success'=>true,'token'=>$token,'user'=>$user]);
        break;
    }
    case 'me': {
        $user = requireAuth();
        unset($user['password_hash']);
        unset($user['token']);
        $user['is_expired'] = isUserExpired($user);
        $user['tier_config'] = getTierConfig($user['tier']);
        jsonResponse(['success'=>true,'user'=>$user]);
        break;

            // Jika private, public diabaikan
            if ($privateType === 'gemini' && $usePrivate) {
                $res = proxyToPrivateGemini($prompt, $images, 'chat', $options);
                if (($res['success'] ?? false) || isset($res['response'])) {
                    jsonResponse($res);
                }
                error_log("[ai_chat] private gemini failed, fallback to openrouter if enabled: ".json_encode($res));
                // fallback to openrouter if gemini fails but openrouter enabled
                if ($useOpenRouter) {
                    $orOptions = $options;
                    if (empty($input['model']) && empty($_POST['model'])) {
                        $orOptions['model'] = $open['text_model'] ?? 'google/gemma-3-27b-it:free';
                        if (!empty($images)) $orOptions['model'] = $open['vision_model'] ?? $open['text_model'];
                    }
                    $res2 = proxyToOpenRouter($prompt, $images, 'chat', $orOptions);
                    if (($res2['success'] ?? false) || isset($res2['response'])) {
                        jsonResponse($res2);
                    }
                }
            } elseif ($privateType === 'openrouter' && $useOpenRouter) {
                $orOptions = $options;
                if (empty($input['model']) && empty($_POST['model'])) {
                    $orOptions['model'] = $open['text_model'] ?? 'google/gemma-3-27b-it:free';
                    if (!empty($images)) $orOptions['model'] = $open['vision_model'] ?? $open['text_model'];
                }
                $res = proxyToOpenRouter($prompt, $images, 'chat', $orOptions);
                if (($res['success'] ?? false) || isset($res['response'])) {
                    jsonResponse($res);
                }
                error_log("[ai_chat] private openrouter failed, fallback to gemini if enabled: ".json_encode($res));
                if ($usePrivate) {
                    $res2 = proxyToPrivateGemini($prompt, $images, 'chat', $options);
                    if (($res2['success'] ?? false) || isset($res2['response'])) {
                        jsonResponse($res2);
                    }
                }
            } else {
                // private selected but chosen type not enabled, try other private type
                if ($usePrivate) {
                    $res = proxyToPrivateGemini($prompt, $images, 'chat', $options);
                    if (($res['success'] ?? false) || isset($res['response'])) jsonResponse($res);
                }
                if ($useOpenRouter) {
                    $orOptions = $options;
                    if (empty($input['model']) && empty($_POST['model'])) {
                        $orOptions['model'] = $open['text_model'] ?? 'google/gemma-3-27b-it:free';
                        if (!empty($images)) $orOptions['model'] = $open['vision_model'] ?? $open['text_model'];
                    }
                    $res = proxyToOpenRouter($prompt, $images, 'chat', $orOptions);
                    if (($res['success'] ?? false) || isset($res['response'])) jsonResponse($res);
                }
            }
            // If private selected, do NOT fallback to public (sesuai request: private maka public diabaikan)
            jsonResponse(['success'=>false,'error'=>'Private API selected but not configured or failed'], 500);
        } else {
            // provider = public, private diabaikan, langsung public
            // Public mode handled below (proxy)
        }
        // If provider=public OR private failed and we allow fallback to public only when provider=public
        if ($provider === 'public') {
            // skip private, go to public
        } else {
            // Legacy fallback for old behavior if provider not set: try private/openrouter then public
            if ($usePrivate) {
                $res = proxyToPrivateGemini($prompt, $images, 'chat', $options);
                if (($res['success'] ?? false) || isset($res['response'])) {
                    jsonResponse($res);
                }
            }
            if ($useOpenRouter) {
                $orOptions = $options;
                if (empty($input['model']) && empty($_POST['model'])) {
                    $orOptions['model'] = $open['text_model'] ?? 'google/gemma-3-27b-it:free';
                    if (!empty($images)) $orOptions['model'] = $open['vision_model'] ?? $open['text_model'];
                }
                $res = proxyToOpenRouter($prompt, $images, 'chat', $orOptions);
                if (($res['success'] ?? false) || isset($res['response'])) {
                    jsonResponse($res);
                }
            }
        }
        // Public proxy mode
        $payload = [];
        $payload['prompt'] = $prompt;
        $tmpFiles = [];
        foreach ($images as $idx=>$dataUrl) {
            if (strpos($dataUrl, 'data:')===0) {
                preg_match('/data:(.*?);base64,(.*)/', $dataUrl, $m);
                $mime = $m[1] ?? 'image/jpeg';
                $ext = strpos($mime,'png')!==false ? 'png' : 'jpg';
                $tmp = tempnam(sys_get_temp_dir(), 'aislide_') . ".$ext";
                file_put_contents($tmp, base64_decode($m[2]));
                $tmpFiles[] = $tmp;
                $payload["images"] = new CURLFile($tmp, $mime, "ref_$idx.$ext");
                break;
            }
        }
        $res = proxyToPublicServer('chat', $payload, true);
        foreach ($tmpFiles as $tf) @unlink($tf);
        jsonResponse($res);
        break;
    }
    case 'ai_generate': {
        $user = requireAuth();
        if (isUserExpired($user)) jsonResponse(['success'=>false,'error'=>'expired','message'=>'Masa berlaku habis.'], 403);
        $instruction = $input['instruction'] ?? $_POST['instruction'] ?? $input['prompt'] ?? '';
        $aspect = $input['aspectRatio'] ?? $_POST['aspectRatio'] ?? $_POST['aspect_ratio'] ?? '16:9';
        $images = [];
        if (isset($input['images']) && is_array($input['images'])) $images = $input['images'];
        elseif (!empty($_FILES)) {
            foreach ($_FILES as $f) {
                if ($f['error']===UPLOAD_ERR_OK) {
                    $mime = mime_content_type($f['tmp_name']);
                    $b64 = base64_encode(file_get_contents($f['tmp_name']));
                    $images[] = "data:$mime;base64,$b64";
                }
            }
        }

        $priv = getPrivateConfig();
        $open = getOpenRouterConfig();
        $usePrivate = !empty($priv['enabled']) && !empty($priv['api_key']);
        $useOpenRouter = !empty($open['enabled']) && !empty($open['api_key']);
        $options = [];
        $options['aspectRatio'] = $aspect;
        if (!empty($input['model'])) $options['model'] = $input['model'];
        elseif (!empty($_POST['model'])) $options['model'] = $_POST['model'];
        if (!empty($input['image_size'])) $options['image_size'] = $input['image_size'];
        if (empty($options['model']) && $usePrivate) {
            $options['model'] = $priv['image_model'] ?? 'gemini-3.1-flash-image';
        }

        if ($usePrivate) {
            $res = proxyToPrivateGemini($instruction, $images, 'generate', $options);
            if (($res['success'] ?? false) && !empty($res['imageUrl'])) {
                jsonResponse($res);
            }
            error_log("[ai_generate] private failed model=".($options['model']??'')." err=".json_encode($res));
        }
        if ($useOpenRouter) {
            $orOptions = $options;
            if (empty($input['model']) && empty($_POST['model'])) {
                $orOptions['model'] = $open['image_model'] ?? 'openai/gpt-image-1-mini';
            }
            $res = proxyToOpenRouter($instruction, $images, 'generate', $orOptions);
            if (($res['success'] ?? false) && !empty($res['imageUrl'])) {
                jsonResponse($res);
            }
            error_log("[ai_generate] openrouter failed model=".($orOptions['model']??'')." err=".json_encode($res));
        }
        $payload = [];
        $payload['instruction'] = $instruction;
        $payload['aspectRatio'] = $aspect;
        $tmpFiles = [];
        if (!empty($images)) {
            $dataUrl = $images[0];
            if (strpos($dataUrl, 'data:')===0) {
                preg_match('/data:(.*?);base64,(.*)/', $dataUrl, $m);
                $mime = $m[1] ?? 'image/jpeg';
                $ext = strpos($mime,'png')!==false ? 'png' : 'jpg';
                $tmp = tempnam(sys_get_temp_dir(), 'aislide_') . ".$ext";
                file_put_contents($tmp, base64_decode($m[2]));
                $tmpFiles[] = $tmp;
                $payload['images'] = new CURLFile($tmp, $mime, "ref.$ext");
            }
        }
        $res = proxyToPublicServer('generate', $payload, true);
        foreach ($tmpFiles as $tf) @unlink($tf);
        jsonResponse($res);
        break;
    }
    case 'private_status': {
        $user = requireAuth();
        $priv = getPrivateConfig();
        $open = getOpenRouterConfig();
        $pub = getPublicApiConfig();
        $enabled = !empty($priv['enabled']) && !empty($priv['api_key']);
        $models = $priv['image_models'] ?? [
            'nano_banana' => 'gemini-2.5-flash-image-preview',
            'nano_banana_2' => 'gemini-3.1-flash-image',
            'nano_banana_2_lite' => 'gemini-3.1-flash-lite-image'
        ];
        jsonResponse([
            'success'=>true,
            'enabled'=>$enabled,
            'image_models'=>$models,
            'default_image_model'=>$priv['image_model']??'gemini-3.1-flash-image',
            'text_model'=>$priv['text_model']??'gemini-3.1-flash-lite',
            'image_size'=>'1K',
            'image_aspect'=>'16:9',
            'openrouter_enabled'=>!empty($open['enabled']) && !empty($open['api_key']),
            'openrouter_text_model'=>$open['text_model']??'google/gemma-3-27b-it:free',
            'openrouter_image_model'=>$open['image_model']??'openai/gpt-image-1-mini',
            'public_mode'=>$pub['mode']??'proxy'
        ]);
        break;
    }
    case 'public_direct_config': {
        $user = requireAuth();
        $auth = $config['public_auth'] ?? [];
        $apiKey = $auth['api_key'] ?? '';
        $jwt = generatePublicJWT();
        $ttl = intval($auth['ttl_min'] ?? 30) * 60;
        $pub = getPublicApiConfig();
        $open = getOpenRouterConfig();
        $priv = getPrivateConfig();
        $aiProvider = getAiProviderConfig();
        jsonResponse([
            'success'=>true,
            'apiKey'=>$apiKey,
            'api_key'=>$apiKey,
            'token'=>$jwt,
            'expires_in'=>$ttl,
            'exp'=>time() + $ttl,
            'servers'=>$config['public_servers'],
            'mode'=>$pub['mode']??'proxy',
            'public_mode'=>$pub['mode']??'proxy',
            'direct_retry'=>$pub['direct_retry']??3,
            'clone_headers'=>!empty($pub['clone_headers']),
            'private_enabled'=>!empty($priv['enabled']),
            'openrouter_enabled'=>!empty($open['enabled']),
            'ai_provider'=>$aiProvider['provider']??'private',
            'private_type'=>$aiProvider['private_type']??'gemini',
            'ai_provider_config'=>$aiProvider
        ]);
        break;
    }
    default:
        jsonResponse(['success'=>false,'error'=>'unknown action: '.$action], 400);
}
